#if UNITY_EDITOR
using UnityEditor;
sealed class LocalC3Import:AssetPostprocessor{void OnPreprocessModel(){if(!assetPath.Contains("LocalC3"))return;var m=(ModelImporter)assetImporter;m.importNormals=ModelImporterNormals.Import;m.importTangents=ModelImporterTangents.None;m.isReadable=true;m.meshCompression=ModelImporterMeshCompression.Off;m.optimizeMeshVertices=false;m.optimizeMeshPolygons=false;m.weldVertices=false;m.importBlendShapes=false;
 if(assetPath.Contains("/W_")){m.animationType=ModelImporterAnimationType.None;m.importAnimation=false;}
 else if(assetPath.Contains("/H_")){m.animationType=ModelImporterAnimationType.Human;m.avatarSetup=ModelImporterAvatarSetup.CreateFromThisModel;m.importAnimation=false;}
 else{m.animationType=ModelImporterAnimationType.Legacy;m.importAnimation=true;}}}
#endif
