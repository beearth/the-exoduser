 if(assetPath.Contains("/W_"))m.animationType=ModelImporterAnimationType.None;
 else if(assetPath.Contains("/H_")){m.animationType=ModelImporterAnimationType.Human;m.avatarSetup=ModelImporterAvatarSetup.CreateFromThisModel;}
 else m.animationType=ModelImporterAnimationType.Generic;}}
