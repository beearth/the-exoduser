#if UNITY_EDITOR
using UnityEditor;
sealed class LocalMxImport:AssetPostprocessor{void OnPreprocessModel(){var m=(ModelImporter)assetImporter;
 if(assetPath.Contains("Resources/LocalC3")){m.importNormals=ModelImporterNormals.Import;m.importTangents=ModelImporterTangents.None;m.isReadable=true;m.meshCompression=ModelImporterMeshCompression.Off;m.optimizeMeshVertices=false;m.optimizeMeshPolygons=false;m.weldVertices=false;m.importBlendShapes=false;m.importAnimation=false;
  if(assetPath.Contains("/W_"))m.animationType=ModelImporterAnimationType.None;else if(assetPath.Contains("/H_")){m.animationType=ModelImporterAnimationType.Human;m.avatarSetup=ModelImporterAvatarSetup.CreateFromThisModel;}else m.animationType=ModelImporterAnimationType.Generic;return;}
 if(!assetPath.Contains("Resources/LocalMx"))return;
 if(assetPath.Contains("/XBot")){m.animationType=ModelImporterAnimationType.Human;m.avatarSetup=ModelImporterAvatarSetup.CreateFromThisModel;m.importAnimation=false;m.isReadable=false;}
 else{m.animationType=ModelImporterAnimationType.Legacy;m.importAnimation=true;m.animationCompression=ModelImporterAnimationCompression.Off;}}}
#endif
