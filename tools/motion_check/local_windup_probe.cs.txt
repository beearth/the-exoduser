 public sealed partial class NativeGame {
  [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
  static void LocalWindupBoot(){
   var a=Environment.GetCommandLineArgs();if(Array.IndexOf(a,"--local-windup-log")<0&&Array.IndexOf(a,"--local-windup-video")<0)return;
   var g=new GameObject("LocalWindupBoot");DontDestroyOnLoad(g);g.AddComponent<LocalWindupBooter>();
  }
  sealed class LocalWindupBooter:MonoBehaviour{
   IEnumerator Start(){while(Instance==null)yield return null;yield return null;yield return null;Instance.StartCoroutine(Instance.LocalWindupRun());}
  }
  Keyboard lwKb;Mouse lwMs;StringBuilder lwLog;
  void LwTick(string scen,int i,bool lmb){
   var s=Simulation;var p=s.Player;p.Iframes=100000;p.Hp=p.MaxHp;p.Stamina=p.MaxStamina;p.Mana=p.MaxMana;
   var screen=HudCamera.WorldToScreenPoint(NativeRemodel3D.Active?new Vector3(NativeRemodel3D.Point(p.X+300,p.Y).x,NativeRemodel3D.Point(p.X,p.Y).y,NativeRemodel3D.Point(p.X+300,p.Y).z):NativeRemodel3D.Point(p.X+300,p.Y));
   var ms=new MouseState{position=new Vector2(screen.x,screen.y)};if(lmb)ms=ms.WithButton(MouseButton.Left,true);
   lwKb.MakeCurrent();lwMs.MakeCurrent();InputState.Change(lwKb,new KeyboardState());InputState.Change(lwMs,ms);
   foreach(var e in s.Enemies){e.X=p.X+(float)Math.Cos(p.Facing)*40;e.Y=p.Y+(float)Math.Sin(p.Facing)*40;e.Hp=Math.Max(e.Hp,50000000);}
   Running=true;FixedUpdate();Running=false;UpdateViews();
   int act=0,dmg=0;foreach(var q in s.Projectiles)if(q.Active){act++;dmg+=q.Damage;}
   double hp=0;foreach(var e in s.Enemies)hp+=e.Hp+e.Shield;
   lwLog.Append(scen).Append(',').Append(i).Append(',').Append(s.TickCount).Append(',').Append(lmb?1:0).Append(',').Append(p.State).Append(',').Append(p.StateTicks).Append(',').Append(s.KiComboStep).Append(',').Append(s.KiComboTicks).Append(',').Append(p.KiChargeActive?1:0).Append(',').Append(p.KiChargeTicks).Append(',').Append(act).Append(',').Append(dmg).Append(',').Append(hp.ToString("F0")).Append(',').Append(p.Facing.ToString("F3")).Append(',').Append(p.X.ToString("F1")).Append(',').Append(p.Y.ToString("F1")).Append(',').Append(s.Enemies.Count).Append(',').Append(s.Enemies.Count>0&&s.Enemies[0].Alive?1:0).Append(',').Append(s.Enemies.Count>0?s.Enemies[0].HurtTicks:0).Append('\n');
  }
  IEnumerator LocalWindupRun(){
   string output=Argument("-output");if(string.IsNullOrEmpty(output)||string.IsNullOrEmpty(Argument("--unity-qa-sandbox"))){Application.Quit(2);yield break;}
   bool video=Array.IndexOf(Environment.GetCommandLineArgs(),"--local-windup-video")>=0;
   Directory.CreateDirectory(output);AudioListener.volume=0;AudioListener.pause=true;Application.runInBackground=true;smokeActive=true;pauseSmoke=true;
   Screen.SetResolution(1920,1080,FullScreenMode.Windowed);yield return new WaitForSecondsRealtime(.4f);
   lobby.Flow.ShowLobby();lobby.Flow.SelectPreview();if(!lobby.Flow.EnterTest()){File.WriteAllText(Path.Combine(output,"lw.txt"),"ENTER_FAIL\n");Application.Quit(2);yield break;}
   EndFirstPractice(true);frontendEntryInputFence=false;menu=false;Running=false;SetGameFocus(true);
   var s=Simulation;var p=s.Player;
   float[][] cands={new[]{3450f,7900f},new[]{3920f,7300f},new[]{4200f,7460f},new[]{3600f,7600f},new[]{3800f,7700f},new[]{4000f,7100f},new[]{3300f,7500f},new[]{3700f,7200f}};
   p.X=cands[0][0];p.Y=cands[0][1];string scan="";foreach(var c in cands){bool free=true;for(int k=1;k<=26&&free;k++){float qx=c[0]+k*14,qy=c[1];if(s.IsProjectileWall!=null?s.IsProjectileWall(qx,qy):!Map.CanMove(qx,qy,1))free=false;}scan+=c[0]+":"+c[1]+"="+free+" ";if(free){p.X=c[0];p.Y=c[1];break;}}
   File.WriteAllText(Path.Combine(output,"scan.txt"),scan);s.Config.EnemyAiEnabled=false;s.Enemies.Clear();s.SpawnHoles=null;
   float deadline=Time.realtimeSinceStartup+40;while(!Remodel3DReady&&Time.realtimeSinceStartup<deadline){UpdateViews();yield return null;}
   if(RigPixelRequested){float rd=Time.realtimeSinceStartup+20;while((rigPixelHero==null||!rigPixelHero.Ready)&&Time.realtimeSinceStartup<rd){UpdateViews();yield return null;}}
   yield return new WaitForSecondsRealtime(1);
   lwKb=InputSystem.AddDevice<Keyboard>();lwMs=InputSystem.AddDevice<Mouse>();
   lwLog=new StringBuilder("scen,i,tick,lmb,state,st,kiStep,kiTicks,charging,charge,proj,projDmg,enemyHp,facing,px,py,enemies,alive,hurt\n");
   string info="active="+s.Config.ActiveAttack+" weaponSpeed="+s.Config.WeaponSpeed+" asMul="+s.Config.AttackSpeedMultiplier+" rig="+RigPixelRequested+" remodel="+Remodel3DReady+" heroReady="+(heroView!=null&&heroView.Ready)+"\n";
   File.WriteAllText(Path.Combine(output,"lw.txt"),info);
   var dummy=s.SpawnNormalEnemy(p.X+50,p.Y);dummy.Hp=dummy.MaxHp=100000000;
   for(int i=0;i<30;i++){LwTick("settle",i,false);yield return null;}
   if(video){
    Time.captureFramerate=60;string dir=Path.Combine(output,"frames");Directory.CreateDirectory(dir);
    for(int f=0;f<600;f++){
     LwTick("video",f,f>=60&&(f-60)%120<3);
     yield return new WaitForEndOfFrame();
     var tex=ScreenCapture.CaptureScreenshotAsTexture();File.WriteAllBytes(Path.Combine(dir,"f"+f.ToString("0000")+".jpg"),tex.EncodeToJPG(92));Destroy(tex);
     if(f%60==0)File.WriteAllText(Path.Combine(output,"video-log.csv"),lwLog.ToString());
    }
    Time.captureFramerate=0;File.WriteAllText(Path.Combine(output,"video-log.csv"),lwLog.ToString());
   }else{
    // A: a single 3-tick click.  B: three taps chained, each when back to idle.  C: 1, 2, then the third held 130 ticks and released.  Rest 180 ticks between (combo window 120).
    int rest=180;
    for(int r=0;r<3;r++){
     string sc="A"+r;for(int i=0;i<3;i++){LwTick(sc,i,true);yield return null;}for(int i=3;i<60+rest;i++){LwTick(sc,i,false);yield return null;}
    }
    foreach(string sc in new[]{"B","C"}){
     int i=0;for(int hit=1;hit<=3;hit++){
      int hold=sc=="C"&&hit==3?130:3;for(int h=0;h<hold;h++){LwTick(sc,i++,true);yield return null;}
      int guard=0;do{LwTick(sc,i++,false);yield return null;}while(p.State!="idle"&&++guard<200);
     }
     for(int h=0;h<rest;h++){LwTick(sc,i++,false);yield return null;}
    }
    File.WriteAllText(Path.Combine(output,"tick-log.csv"),lwLog.ToString());
   }
   File.AppendAllText(Path.Combine(output,"lw.txt"),"DONE\n");Application.Quit(0);
  }
 }
